Private Const c_Projects = "202406"
Private Const c_User = "Master"
Private Const c_Pass = ""

Private Const Alamat = "D:\Stanley\Tugas IFRS17\ Reserving Macro Automation\\" & ValDate & "\Master"
Private Const OriDev = 59
Private Const OutVer = "v7"

Sub CompileToResQ()
    SwitchOff (True)
    
    Application.DisplayAlerts = False
    Application.ScreenUpdating = False
    
    Dim rsSheetsG As ADODB.Recordset
    Dim rsSheetsN As ADODB.Recordset
    Dim rsSheetsNX As ADODB.Recordset
    Dim resQ As New ResQApplication
    Dim projectResQ As xResQProject
    Dim reservingClassResQ As xReservingClass
    Dim linkRow As Integer
    Dim resRow As Integer
    Dim lastRow As Long
    Dim PathName As String
    
    Dim wbEP As Workbook
    Dim wbEPX As Workbook
    Dim wbPCGrs As Workbook
    Dim wbPCNet As Workbook
    Dim wbPCNetX As Workbook
    Dim wbOSGrs As Workbook
    Dim wbOSNet As Workbook
    Dim wbOSNetX As Workbook
    Dim newWorkbook As Workbook

    
    Dim masterSheet As Worksheet
    Dim newSheet As Worksheet
    Dim wsEP As Worksheet
    Dim wsEPX As Worksheet
    Dim wsPCGrs As Worksheet
    Dim wsPCNet As Worksheet
    Dim wsPCNetX As Worksheet
    Dim wsOSGrs As Worksheet
    Dim wsOSNet As Worksheet
    Dim wsOSNetX As Worksheet

    
    Dim SavePath As String
    
    'timer
    Dim startTime As Date
    Dim endTime As Date
    Dim elapsedTime As Double
    Dim formattedTime As String
    startTime = Now
    
    'Looping Folder
    
    PathName = Alamat
    
    linkRow = 1
    resRow = 1
    
    Set wbEP = Workbooks.Open(PathName & "\" & "Earned Premium.xlsx", ReadOnly:=True, UpdateLinks:=0)
    Set wbEPX = Workbooks.Open(PathName & "\" & "Earned Premium - wo XL.xlsx", ReadOnly:=True, UpdateLinks:=0)
    Set wbPCGrs = Workbooks.Open(PathName & "\" & "PaidClaim-Gross.xlsx", ReadOnly:=True, UpdateLinks:=0)
    Set wbPCNet = Workbooks.Open(PathName & "\" & "PaidClaim-Net.xlsx", ReadOnly:=True, UpdateLinks:=0)
    Set wbPCNetX = Workbooks.Open(PathName & "\" & "PaidClaim-Net - wo XL.xlsx", ReadOnly:=True, UpdateLinks:=0)
    Set wbOSGrs = Workbooks.Open(PathName & "\" & "OSClaim-Gross.xlsx", ReadOnly:=True, UpdateLinks:=0)
    Set wbOSNet = Workbooks.Open(PathName & "\" & "OSClaim-Net.xlsx", ReadOnly:=True, UpdateLinks:=0)
    Set wbOSNetX = Workbooks.Open(PathName & "\" & "OSClaim-Net - wo XL.xlsx", ReadOnly:=True, UpdateLinks:=0)
           
    'New WB
    'Set newWorkbook = Workbooks.Add
    
    linkRow = linkRow + 1
    
    'Looping Sheets
    
    'Untuk file-file Gross
    Set rsSheetsG = SheetsListGross
    rsSheetsG.MoveFirst

    If Not rsSheetsG.EOF Then
        Set wsEP = wbEP.Worksheets("EP")

        Call resQ.ConnectByName(c_, c_User, c_Pass)
        Set projectResQ = resQ.Projects(c_Projects)

        Do While Not rsSheetsG.EOF
            Debug.Print rsSheetsG.Fields("ResQ Path")
            Set reservingClassResQ = projectResQ.ReservingClasses(rsSheetsG.Fields("ResQ Path"))
            Set wsPCGrs = wbPCGrs.Worksheets(rsSheetsG.Fields("Sheet Name CL").Value)
            Set wsOSGrs = wbOSGrs.Worksheets(rsSheetsG.Fields("Sheet Name CL").Value)

            'Earned Premium
            For i = 1 To OriDev - 9 '9 karena pemotongan accident year 2009 Q4 sampai 2011 Q4
                reservingClassResQ.Vectors("Ultimate Premiums").ValuesByIndex(i) = wsEP.Cells(i + 10, rsSheetsG.Fields("Column EP").Value)
            Next i

            reservingClassResQ.Vectors("Ultimate Premiums").Save

            'Paid Claims
            reservingClassResQ.Triangles("Paid Claims").Cumulative = False
            For i = 1 To OriDev - 9 'Row TriA

                For j = 1 To OriDev - 9 + 1 - i 'Column TriA

                    reservingClassResQ.Triangles("Paid Claims").ValuesByIndex(i, j) = wsPCGrs.Cells(i + 10, j + 2)

                Next j

            Next i
            reservingClassResQ.Triangles("Paid Claims").Cumulative = True

            reservingClassResQ.Triangles("Paid Claims").Save

            'OS Claims
            For i = 1 To OriDev - 9  'Row TriA

                For j = 1 To OriDev - 9 + 1 - i 'Column TriA

                    reservingClassResQ.Triangles("Outstanding Claims").ValuesByIndex(i, j) = wsOSGrs.Cells(i + 10, j + 2)

                Next j

            Next i

            reservingClassResQ.Triangles("Outstanding Claims").Save

            'Next Sheet

            rsSheetsG.MoveNext
        Loop

        resQ.Disconnect
    End If
    
    'Untuk file-file Net
    Set rsSheetsN = SheetsListNet
    rsSheetsN.MoveFirst

    If Not rsSheetsN.EOF Then
        Set wsEP = wbEP.Worksheets("EP")

        Call resQ.ConnectByName(c_, c_User, c_Pass)
        Set projectResQ = resQ.Projects(c_Projects)

        Do While Not rsSheetsN.EOF
        Debug.Print rsSheetsN.Fields("ResQ Path").Value
            Set reservingClassResQ = projectResQ.ReservingClasses(rsSheetsN.Fields("ResQ Path"))
            Set wsPCNet = wbPCNet.Worksheets(rsSheetsN.Fields("Sheet Name CL").Value)
            Set wsOSNet = wbOSNet.Worksheets(rsSheetsN.Fields("Sheet Name CL").Value)

            'Earned Premium
            For i = 1 To OriDev - 9

                reservingClassResQ.Vectors("Ultimate Premiums").ValuesByIndex(i) = wsEP.Cells(i + 10, rsSheetsN.Fields("Column EP").Value)

            Next i

            reservingClassResQ.Vectors("Ultimate Premiums").Save

            'Paid Claims
            reservingClassResQ.Triangles("Paid Claims").Cumulative = False
            For i = 1 To OriDev - 9  'Row TriA

                For j = 1 To OriDev - 9 + 1 - i 'Column TriA

                    reservingClassResQ.Triangles("Paid Claims").ValuesByIndex(i, j) = wsPCNet.Cells(i + 10, j + 2)

                Next j

            Next i
            reservingClassResQ.Triangles("Paid Claims").Cumulative = True

            reservingClassResQ.Triangles("Paid Claims").Save

            'OS Claims

            For i = 1 To OriDev - 9  'Row TriA

                For j = 1 To OriDev - 9 + 1 - i 'Column TriA

                    reservingClassResQ.Triangles("Outstanding Claims").ValuesByIndex(i, j) = wsOSNet.Cells(i + 10, j + 2)

                Next j

            Next i

            reservingClassResQ.Triangles("Outstanding Claims").Save

            'Next Sheet
            rsSheetsN.MoveNext
        Loop

        resQ.Disconnect
    End If
    
    
    'Untuk file-file Net-wo-XL
    Set rsSheetsNX = SheetsListNetXL
    rsSheetsNX.MoveFirst

    If Not rsSheetsNX.EOF Then
        Set wsEPX = wbEPX.Worksheets("EP")
        
        Call resQ.ConnectByName(c_, c_User, c_Pass)
        Set projectResQ = resQ.Projects(c_Projects)
        
        Do While Not rsSheetsNX.EOF
        Debug.Print rsSheetsNX.Fields("ResQ Path").Value
            Set reservingClassResQ = projectResQ.ReservingClasses(rsSheetsNX.Fields("ResQ Path"))
            Set wsPCNetX = wbPCNetX.Worksheets(rsSheetsNX.Fields("Sheet Name CL").Value)
            Set wsOSNetX = wbOSNetX.Worksheets(rsSheetsNX.Fields("Sheet Name CL").Value)
            
            'Earned Premium
            For i = 1 To OriDev - 9
            
                reservingClassResQ.Vectors("Ultimate Premiums").ValuesByIndex(i) = wsEPX.Cells(i + 10, rsSheetsNX.Fields("Column EP").Value)
                
            Next i
            
            reservingClassResQ.Vectors("Ultimate Premiums").Save
            
            'Paid Claims
            reservingClassResQ.Triangles("Paid Claims").Cumulative = False
            For i = 1 To OriDev - 9  'Row TriA
                
                For j = 1 To OriDev - 9 + 1 - i 'Column TriA
                
                    reservingClassResQ.Triangles("Paid Claims").ValuesByIndex(i, j) = wsPCNetX.Cells(i + 10, j + 2)
                
                Next j
                
            Next i
            reservingClassResQ.Triangles("Paid Claims").Cumulative = True
            
            reservingClassResQ.Triangles("Paid Claims").Save
            
            'OS Claims
            
            For i = 1 To OriDev - 9  'Row TriA
                
                For j = 1 To OriDev - 9 + 1 - i 'Column TriA
                
                    reservingClassResQ.Triangles("Outstanding Claims").ValuesByIndex(i, j) = wsOSNetX.Cells(i + 10, j + 2)
                
                Next j
                
            Next i
            
            reservingClassResQ.Triangles("Outstanding Claims").Save
            
            'Next Sheet
            rsSheetsNX.MoveNext
        Loop
        
        resQ.Disconnect
    End If
    
    'Next Folder
    wbEP.Close (SaveChanges = False)
    wbEPX.Close (SaveChanges = False)
    wbPCGrs.Close (SaveChanges = False)
    wbPCNet.Close (SaveChanges = False)
    wbPCNetX.Close (SaveChanges = False)
    wbOSGrs.Close (SaveChanges = False)
    wbOSNet.Close (SaveChanges = False)
    wbOSNetX.Close (SaveChanges = False)
    
    'Delete Sheet 1
        
    'end time
    endTime = Now
    'elapsed time
    elapsedTime = DateDiff("s", startTime, endTime)
    formattedTime = FormatTime(elapsedTime)
    
    MsgBox "Done!" & vbCrLf & "Running time: " & formattedTime
    
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    
    SwitchOff (False)
End Sub

Private Function FormatTime(ByVal seconds As Double) As String
    Dim hours As Long
    Dim minutes As Long
    Dim remainingSeconds As Long
    
    ' Calculate hours, minutes, and remaining seconds
    hours = seconds \ 3600
    seconds = seconds Mod 3600
    minutes = seconds \ 60
    remainingSeconds = seconds Mod 60
    
    ' Format the time as "hour:minute:second"
    FormatTime = Format(hours, "00") & ":" & Format(minutes, "00") & ":" & Format(remainingSeconds, "00")
End Function
Private Sub SwitchOff(bSwitchOff As Boolean)
    Dim ws As Worksheet
    
    With Application
        If bSwitchOff Then
            'OFF
            .Calculation = xlCalculationManual
            .ScreenUpdating = False
            .EnableAnimations = False
         
            'switch off display pagebreaks for all worksheets
            For Each ws In ThisWorkbook.Worksheets
                ws.DisplayPageBreaks = False
            Next ws
        Else
            'ON
            .Calculation = xlCalculationAutomatic
            .ScreenUpdating = True
            .EnableAnimations = True
        End If
    End With
End Sub
