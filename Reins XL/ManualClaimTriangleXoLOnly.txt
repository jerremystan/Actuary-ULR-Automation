        End If
        
        ratioValue = wsP.Cells(42, 5).Value / wsS.Cells(42, 5).Value
        
        Debug.Print ratioValue
        
        Set ws = Workbooks.Open("D:\Stanley\Tugas IFRS17\ Reserving Macro Automation\\" & ValDate & "\Master\" & FN & ".xlsx", UpdateLinks:=0).Sheets("Fire")
